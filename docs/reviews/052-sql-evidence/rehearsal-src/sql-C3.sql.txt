SELECT t.slug AS tenant_slug, (SELECT count(*) FROM public.capture_documents d WHERE d.tenant_id = t.id) AS documents,
       (SELECT count(*) FROM public.whatsapp_sessions s WHERE s.tenant_id = t.id) AS sessions,
       (SELECT count(*) FROM public.users u WHERE u.tenant_id = t.id) AS users
  FROM public.tenants t WHERE t.slug LIKE 'zz-test-capture-%' ORDER BY t.slug;
