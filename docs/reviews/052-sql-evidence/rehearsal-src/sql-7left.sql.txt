       (SELECT count(*) FROM public.users WHERE full_name LIKE 'ZZTestCapture-f5c4d2cd-%') AS b2a_public_users,

       (SELECT count(*) FROM public.capture_documents d JOIN public.tenants t ON t.id = d.tenant_id WHERE t.slug LIKE 'zz-test-capture-b2a-%') AS b2a_documents,
