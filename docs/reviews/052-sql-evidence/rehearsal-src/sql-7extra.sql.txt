SELECT left(u.full_name, 24) AS full_name_prefix, u.role, u.status, u.created_at, t.slug AS tenant_slug, (u.auth_id IS NOT NULL) AS has_auth_id, (u.tenant_id IS NULL) AS no_tenant
  FROM public.users u LEFT JOIN public.tenants t ON t.id = u.tenant_id
 WHERE u.created_at > '2026-11-13 16:00:00+00' ORDER BY u.created_at;
