CREATE FUNCTION public.trg_dil_before_insert()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  v_status   text;
  v_type     text;
  v_auto_inv uuid;
BEGIN
  -- B3: first statement.
  IF current_setting('transaction_isolation') <> 'read committed' THEN
    RAISE EXCEPTION USING ERRCODE = 'QE014', MESSAGE = 'E14';
  END IF;

  -- B2: a forged INSERT cannot carry unlink or snapshot values.
  IF NEW.unlinked_at IS NOT NULL OR NEW.unlinked_by IS NOT NULL OR NEW.unlink_reason IS NOT NULL
     OR NEW.superseded_from_status IS NOT NULL OR NEW.superseded_approval_by IS NOT NULL
     OR NEW.superseded_approval_at IS NOT NULL THEN
    RAISE EXCEPTION USING ERRCODE = 'QE010', MESSAGE = 'E10';
  END IF;

  -- F4-1 (per Aravind 11 Oct): linked_at is stamped by the database, never
  -- trusted from NEW (it feeds the D-12 clock and the unlinked_at >= linked_at CHECK).
  NEW.linked_at := now();
  -- S-4 (SQL review r3, per Aravind 12 Oct): created_at stamped by the database too.
  NEW.created_at := now();

  -- Lock level 1: the invoice document row, FOR NO KEY UPDATE (R3).
  PERFORM 1 FROM public.capture_documents d WHERE d.id = NEW.invoice_document_id FOR NO KEY UPDATE;

  -- A28 XOR, written with <> so a later auto_match source needs no edit.
  IF EXISTS (
    SELECT 1 FROM public.delivery_invoice_links l
    WHERE l.invoice_document_id = NEW.invoice_document_id
      AND l.unlinked_at IS NULL
      AND (l.link_source = 'created_from_invoice') <> (NEW.link_source = 'created_from_invoice')
  ) THEN
    IF NEW.link_source = 'pm_manual' THEN
      RAISE EXCEPTION USING ERRCODE = 'QE001', MESSAGE = 'E1';
    ELSE
      RAISE EXCEPTION USING ERRCODE = 'QE002', MESSAGE = 'E2';
    END IF;
  END IF;

  -- TR2 (same function, fixed order): FOR SHARE on the target delivery.
  SELECT d.status, d.delivery_type, d.created_from_invoice_id
    INTO v_status, v_type, v_auto_inv
    FROM public.deliveries d
   WHERE d.id = NEW.delivery_id
     FOR SHARE;
  IF FOUND THEN
    IF NEW.link_source = 'pm_manual' THEN
      IF v_auto_inv IS NOT NULL THEN
        RAISE EXCEPTION USING ERRCODE = 'QE004', MESSAGE = 'E4';
      ELSIF v_type = 'internal_transfer' THEN
        RAISE EXCEPTION USING ERRCODE = 'QE003', MESSAGE = 'E3';
      ELSIF v_status IN ('rejected', 'cancelled') THEN
        RAISE EXCEPTION USING ERRCODE = 'QE005', MESSAGE = 'E5';
      END IF;
    ELSIF v_status IN ('rejected', 'cancelled') THEN
      -- B-2 (SQL review r1): a self-link is also refused onto a CANCELLED
      -- delivery. Paths 2 and 4 restore the delivery BEFORE inserting the
      -- new self-link, so this turns that header rule into one the trigger enforces.
      RAISE EXCEPTION USING ERRCODE = 'QE005', MESSAGE = 'E5';
    END IF;
  END IF;

  RETURN NEW;
END;
$$;

CREATE FUNCTION public.trg_dil_before_update()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  v_status  text;
  v_appr_by uuid;
  v_appr_at timestamptz;
BEGIN
  -- B3: first statement.
  IF current_setting('transaction_isolation') <> 'read committed' THEN
    RAISE EXCEPTION USING ERRCODE = 'QE014', MESSAGE = 'E14';
  END IF;

  -- TR3 does not lock the delivery: it runs at lock level 3 (R3).
  -- Identity columns never change.
  IF NEW.id IS DISTINCT FROM OLD.id
     OR NEW.created_at IS DISTINCT FROM OLD.created_at
     OR NEW.tenant_id IS DISTINCT FROM OLD.tenant_id
     OR NEW.project_id IS DISTINCT FROM OLD.project_id
     OR NEW.delivery_id IS DISTINCT FROM OLD.delivery_id
     OR NEW.invoice_document_id IS DISTINCT FROM OLD.invoice_document_id
     OR NEW.doc_kind IS DISTINCT FROM OLD.doc_kind
     OR NEW.link_source IS DISTINCT FROM OLD.link_source
     OR NEW.linked_by IS DISTINCT FROM OLD.linked_by
     OR NEW.linked_at IS DISTINCT FROM OLD.linked_at THEN
    RAISE EXCEPTION USING ERRCODE = 'QE010', MESSAGE = 'E10';
  END IF;

  -- An already-unlinked row is final: nothing set may change, unlinked_at
  -- can never be cleared.
  IF OLD.unlinked_at IS NOT NULL THEN
    IF NEW.unlinked_at IS DISTINCT FROM OLD.unlinked_at
       OR NEW.unlinked_by IS DISTINCT FROM OLD.unlinked_by
       OR NEW.unlink_reason IS DISTINCT FROM OLD.unlink_reason
       OR NEW.superseded_from_status IS DISTINCT FROM OLD.superseded_from_status
       OR NEW.superseded_approval_by IS DISTINCT FROM OLD.superseded_approval_by
       OR NEW.superseded_approval_at IS DISTINCT FROM OLD.superseded_approval_at THEN
      RAISE EXCEPTION USING ERRCODE = 'QE010', MESSAGE = 'E10';
    END IF;
    RETURN NEW;
  END IF;

  -- The row is active. No unlink in this UPDATE: the snapshot columns are
  -- forced NULL, never trusted from NEW (B2).
  IF NEW.unlinked_at IS NULL AND NEW.unlinked_by IS NULL AND NEW.unlink_reason IS NULL THEN
    NEW.superseded_from_status := NULL;
    NEW.superseded_approval_by := NULL;
    NEW.superseded_approval_at := NULL;
    RETURN NEW;
  END IF;

  -- The only allowed change: NULL -> value on unlinked_by + unlinked_at +
  -- unlink_reason TOGETHER.
  IF NEW.unlinked_at IS NULL OR NEW.unlinked_by IS NULL OR NEW.unlink_reason IS NULL THEN
    RAISE EXCEPTION USING ERRCODE = 'QE010', MESSAGE = 'E10';
  END IF;

  -- S-3 (SQL review r2, Option A per Aravind 11 Oct): the unlink time is
  -- stamped by the database; the caller's value only signals the unlink and is
  -- overwritten. (linked_at is NOT touched here: see the package, OPEN item.)
  NEW.unlinked_at := now();

  -- Overwrite the snapshot from the delivery row (B2). Only a self-link
  -- unlinked as superseded_by_link records the delivery's pre-cancel state.
  IF OLD.link_source = 'created_from_invoice' AND NEW.unlink_reason = 'superseded_by_link' THEN
    SELECT d.status, d.approved_by, d.approved_at
      INTO v_status, v_appr_by, v_appr_at
      FROM public.deliveries d
     WHERE d.id = OLD.delivery_id;
    IF NOT FOUND OR NOT COALESCE(v_status IN ('incomplete', 'awaiting_pm', 'approved'), false) THEN
      RAISE EXCEPTION USING ERRCODE = 'QE010', MESSAGE = 'E10';
    END IF;
    NEW.superseded_from_status := v_status;
    IF v_status = 'approved' THEN
      NEW.superseded_approval_by := v_appr_by;
      NEW.superseded_approval_at := v_appr_at;
    ELSE
      NEW.superseded_approval_by := NULL;
      NEW.superseded_approval_at := NULL;
    END IF;
  ELSE
    NEW.superseded_from_status := NULL;
    NEW.superseded_approval_by := NULL;
    NEW.superseded_approval_at := NULL;
  END IF;

  RETURN NEW;
END;
$$;

CREATE FUNCTION public.trg_deliveries_before_update()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
BEGIN
  -- B3: first statement.
  IF current_setting('transaction_isolation') <> 'read committed' THEN
    RAISE EXCEPTION USING ERRCODE = 'QE014', MESSAGE = 'E14';
  END IF;

  -- TR4a: immutable columns (S-1, SQL review r1: id and created_at added --
  -- every ordering CHECK is measured against created_at).
  IF NEW.id IS DISTINCT FROM OLD.id
     OR NEW.created_at IS DISTINCT FROM OLD.created_at
     OR NEW.created_from_invoice_id IS DISTINCT FROM OLD.created_from_invoice_id
     OR NEW.dc_document_id IS DISTINCT FROM OLD.dc_document_id
     OR NEW.project_id IS DISTINCT FROM OLD.project_id
     OR NEW.tenant_id IS DISTINCT FROM OLD.tenant_id
     OR NEW.reported_by IS DISTINCT FROM OLD.reported_by
     OR NEW.received_at IS DISTINCT FROM OLD.received_at THEN
    RAISE EXCEPTION USING ERRCODE = 'QE009', MESSAGE = 'E9';
  END IF;

  -- Self-enforcing lock contract (R3): for an automatic delivery the first
  -- lock action is the invoice document, FOR NO KEY UPDATE. A correct caller
  -- already holds it (no-op); an out-of-order caller deadlocks (40P01)
  -- instead of racing silently. The unlocked read of created_from_invoice_id
  -- that finds the invoice is safe ONLY because TR4a above makes that column
  -- immutable.
  IF NEW.created_from_invoice_id IS NOT NULL THEN
    PERFORM 1 FROM public.capture_documents d WHERE d.id = NEW.created_from_invoice_id FOR NO KEY UPDATE;
  END IF;

  -- O-3 (closed per Aravind 11 Oct): an AUTOMATIC delivery can never be an
  -- internal transfer. Raised BEFORE the active-link check below, so a PM
  -- retyping an automatic delivery gets E7, not E6 ("Unlink it first" -- which
  -- cannot be done for a self-link). The CHECK deliveries_auto_not_internal_chk
  -- stays as the backstop (it also covers an INSERT).
  IF NEW.delivery_type = 'internal_transfer' AND OLD.delivery_type IS DISTINCT FROM 'internal_transfer'
     AND NEW.created_from_invoice_id IS NOT NULL THEN
    RAISE EXCEPTION USING ERRCODE = 'QE007', MESSAGE = 'E7';
  END IF;

  -- TR4b: no retype to internal_transfer while an active link exists.
  IF NEW.delivery_type = 'internal_transfer' AND OLD.delivery_type IS DISTINCT FROM 'internal_transfer' THEN
    IF EXISTS (SELECT 1 FROM public.delivery_invoice_links l WHERE l.delivery_id = NEW.id AND l.unlinked_at IS NULL) THEN
      RAISE EXCEPTION USING ERRCODE = 'QE006', MESSAGE = 'E6';
    END IF;
  END IF;

  -- TR4c (1): status transitions. Nothing leaves rejected.
  IF NEW.status IS DISTINCT FROM OLD.status THEN
    IF NOT (
         (OLD.status = 'incomplete'  AND NEW.status IN ('awaiting_pm', 'approved', 'rejected', 'cancelled'))
      OR (OLD.status = 'awaiting_pm' AND NEW.status IN ('approved', 'rejected', 'cancelled'))
      OR (OLD.status = 'approved'    AND NEW.status = 'cancelled')
      OR (OLD.status = 'cancelled'   AND NEW.status IN ('incomplete', 'awaiting_pm'))
    ) THEN
      RAISE EXCEPTION USING ERRCODE = 'QE011', MESSAGE = 'E11';
    END IF;
    -- TR4c (2)(3): B1 and C-36 -- no move to rejected/cancelled while ANY
    -- active link exists (DC or automatic): the paths unlink first.
    IF NEW.status IN ('rejected', 'cancelled') THEN
      IF EXISTS (SELECT 1 FROM public.delivery_invoice_links l WHERE l.delivery_id = NEW.id AND l.unlinked_at IS NULL) THEN
        RAISE EXCEPTION USING ERRCODE = 'QE013', MESSAGE = 'E13';
      END IF;
    END IF;
  END IF;

  -- TR4c (4): approved means locked (D-23). While OLD is approved or
  -- cancelled these four content fields are immutable.
  IF OLD.status IN ('approved', 'cancelled') THEN
    IF NEW.delivery_type IS DISTINCT FROM OLD.delivery_type
       OR NEW.vehicle_number IS DISTINCT FROM OLD.vehicle_number
       OR NEW.dc_number IS DISTINCT FROM OLD.dc_number
       OR NEW.dc_date IS DISTINCT FROM OLD.dc_date THEN
      RAISE EXCEPTION USING ERRCODE = 'QE011', MESSAGE = 'E11';
    END IF;
  END IF;

  -- B-1 (SQL review r1): approval attribution cannot be rewritten after the
  -- fact. Each group may change ONLY on its own status transition; any other
  -- IS DISTINCT FROM on these columns raises E11.
  -- approved_by/approved_at: NULL -> value only on a move INTO approved;
  -- value -> NULL only on restore (cancelled -> incomplete/awaiting_pm).
  IF NEW.approved_by IS DISTINCT FROM OLD.approved_by
     OR NEW.approved_at IS DISTINCT FROM OLD.approved_at THEN
    IF NOT (
         (OLD.approved_by IS NULL AND OLD.approved_at IS NULL
          AND NEW.status = 'approved' AND OLD.status IS DISTINCT FROM 'approved')
      OR (NEW.approved_by IS NULL AND NEW.approved_at IS NULL
          AND OLD.status = 'cancelled' AND NEW.status IN ('incomplete', 'awaiting_pm'))
    ) THEN
      RAISE EXCEPTION USING ERRCODE = 'QE011', MESSAGE = 'E11';
    END IF;
  END IF;
  -- rejected_by/rejected_at/reject_reason: set only on a move INTO rejected.
  IF NEW.rejected_by IS DISTINCT FROM OLD.rejected_by
     OR NEW.rejected_at IS DISTINCT FROM OLD.rejected_at
     OR NEW.reject_reason IS DISTINCT FROM OLD.reject_reason THEN
    IF NOT (NEW.status = 'rejected' AND OLD.status IS DISTINCT FROM 'rejected') THEN
      RAISE EXCEPTION USING ERRCODE = 'QE011', MESSAGE = 'E11';
    END IF;
  END IF;
  -- cancelled_at/cancel_reason: set on a move INTO cancelled; cleared on restore.
  IF NEW.cancelled_at IS DISTINCT FROM OLD.cancelled_at
     OR NEW.cancel_reason IS DISTINCT FROM OLD.cancel_reason THEN
    IF NOT (
         (NEW.status = 'cancelled' AND OLD.status IS DISTINCT FROM 'cancelled')
      OR (NEW.cancelled_at IS NULL AND NEW.cancel_reason IS NULL
          AND OLD.status = 'cancelled' AND NEW.status IN ('incomplete', 'awaiting_pm'))
    ) THEN
      RAISE EXCEPTION USING ERRCODE = 'QE011', MESSAGE = 'E11';
    END IF;
  END IF;

  -- S-3 (SQL review r2, Option A per Aravind 11 Oct): the DATABASE stamps the
  -- time at each allowed transition; any caller-supplied value is overwritten
  -- (the same "wrong stamp impossible, not merely uncaught" rule as 043's
  -- generated expires_at and B2's snapshot overwrite). Clearing on restore is
  -- left as validated above. received_at stays caller-supplied (D-25).
  IF NEW.status = 'approved' AND OLD.status IS DISTINCT FROM 'approved' THEN
    NEW.approved_at := now();
  END IF;
  IF NEW.status = 'rejected' AND OLD.status IS DISTINCT FROM 'rejected' THEN
    NEW.rejected_at := now();
  END IF;
  IF NEW.status = 'cancelled' AND OLD.status IS DISTINCT FROM 'cancelled' THEN
    NEW.cancelled_at := now();
  END IF;

  RETURN NEW;
END;
$$;

AS $$
BEGIN
  -- B3: first statement.
  IF current_setting('transaction_isolation') <> 'read committed' THEN
    RAISE EXCEPTION USING ERRCODE = 'QE014', MESSAGE = 'E14';
  END IF;

  -- S-1 (SQL review r1): id and created_at added to the immutable set.
  IF NEW.id IS DISTINCT FROM OLD.id
     OR NEW.created_at IS DISTINCT FROM OLD.created_at
     OR NEW.kind IS DISTINCT FROM OLD.kind

     OR NEW.source_message_sid IS DISTINCT FROM OLD.source_message_sid THEN

  END IF;

  -- The acknowledgement is settable once; the note is settable once.
  IF OLD.engineer_ack IS NOT NULL THEN
    IF NEW.engineer_ack IS DISTINCT FROM OLD.engineer_ack
       OR NEW.engineer_ack_at IS DISTINCT FROM OLD.engineer_ack_at THEN
      RAISE EXCEPTION USING ERRCODE = 'QE011', MESSAGE = 'E11';
    END IF;
  END IF;
  IF OLD.engineer_ack_note IS NOT NULL THEN
    IF NEW.engineer_ack_note IS DISTINCT FROM OLD.engineer_ack_note THEN
      RAISE EXCEPTION USING ERRCODE = 'QE011', MESSAGE = 'E11';
    END IF;
  END IF;

  -- S-3 (SQL review r2, Option A per Aravind 11 Oct): the acknowledgement time
  -- is stamped by the database when the ack is first set; a caller-supplied
  -- value is overwritten. It drives the D-12 clock.
  IF OLD.engineer_ack IS NULL AND NEW.engineer_ack IS NOT NULL THEN
    NEW.engineer_ack_at := now();
  END IF;

  RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION public.quoco_test_capture_insert(p_table text, p_row jsonb)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  v_tenant uuid;
  v_cols   text;
  v_out    jsonb;
BEGIN
  IF p_table IS NULL OR p_table NOT IN ('deliveries', 'capture_documents', 'capture_files', 'delivery_invoice_links') THEN
    RAISE EXCEPTION 'HELPER_TABLE_NOT_ALLOWED';
  END IF;
  IF jsonb_typeof(p_row) IS DISTINCT FROM 'object' OR p_row = '{}'::jsonb THEN
    RAISE EXCEPTION 'HELPER_ROW_NOT_OBJECT';
  END IF;
  BEGIN
    v_tenant := (p_row ->> 'tenant_id')::uuid;
  EXCEPTION WHEN others THEN
    v_tenant := NULL;
  END;
  IF NOT public.quoco_test_capture_fixture_tenant_ok(v_tenant) THEN
    RAISE EXCEPTION 'HELPER_TENANT_NOT_FIXTURE';
  END IF;
  SELECT string_agg(format('%I', k), ', ' ORDER BY k) INTO v_cols FROM jsonb_object_keys(p_row) AS k;
  EXECUTE format(
    'INSERT INTO public.%1$I (%2$s) SELECT %2$s FROM jsonb_populate_record(NULL::public.%1$I, $1) RETURNING to_jsonb(%1$I)',
    p_table, v_cols
  ) INTO v_out USING p_row;
  RETURN v_out;
END;
$$;

CREATE OR REPLACE FUNCTION public.quoco_test_capture_update(p_table text, p_id uuid, p_set jsonb)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  v_tenant uuid;
  v_sets   text;
  v_out    jsonb;
BEGIN
  IF p_table IS NULL OR p_table NOT IN ('deliveries', 'capture_documents', 'capture_files', 'delivery_invoice_links') THEN
    RAISE EXCEPTION 'HELPER_TABLE_NOT_ALLOWED';
  END IF;
  IF jsonb_typeof(p_set) IS DISTINCT FROM 'object' OR p_set = '{}'::jsonb THEN
    RAISE EXCEPTION 'HELPER_ROW_NOT_OBJECT';
  END IF;
  EXECUTE format('SELECT tenant_id FROM public.%I WHERE id = $1', p_table) INTO v_tenant USING p_id;
  IF NOT public.quoco_test_capture_fixture_tenant_ok(v_tenant) THEN
    RAISE EXCEPTION 'HELPER_TENANT_NOT_FIXTURE';
  END IF;
  SELECT string_agg(format('%I = (jsonb_populate_record(NULL::public.%I, $2)).%I', k, p_table, k), ', ' ORDER BY k)
    INTO v_sets FROM jsonb_object_keys(p_set) AS k;
  EXECUTE format('UPDATE public.%1$I SET %2$s WHERE id = $1 RETURNING to_jsonb(%1$I)', p_table, v_sets)
    INTO v_out USING p_id, p_set;
  RETURN v_out;
END;

CREATE OR REPLACE FUNCTION public.quoco_test_capture_acl()
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  v_t   text;
  v_r   text;
  v_p   text;
  v_f   text;
  v_tbl jsonb := '{}'::jsonb;
  v_fn  jsonb := '{}'::jsonb;
  v_row jsonb;
  v_cell jsonb;
BEGIN
  FOREACH v_t IN ARRAY ARRAY['capture_documents', 'capture_files', 'deliveries', 'delivery_invoice_links'] LOOP
    v_row := '{}'::jsonb;
    FOREACH v_r IN ARRAY ARRAY['anon', 'authenticated', 'service_role'] LOOP
      v_cell := '{}'::jsonb;
      FOREACH v_p IN ARRAY ARRAY['SELECT', 'INSERT', 'UPDATE', 'DELETE', 'TRUNCATE', 'REFERENCES', 'TRIGGER', 'MAINTAIN'] LOOP
        v_cell := v_cell || jsonb_build_object(v_p, has_table_privilege(v_r, 'public.' || v_t, v_p));
      END LOOP;
      v_row := v_row || jsonb_build_object(v_r, v_cell);
    END LOOP;
    v_tbl := v_tbl || jsonb_build_object(v_t, v_row);
  END LOOP;
  FOREACH v_f IN ARRAY ARRAY['trg_dil_before_insert', 'trg_dil_before_update', 'trg_deliveries_before_update', 'trg_capture_documents_before_update'] LOOP
    SELECT jsonb_build_object(
             'exists', true,
             'prosecdef', p.prosecdef,
             'proconfig', to_jsonb(p.proconfig),
             'owner', pg_get_userbyid(p.proowner)::text,
             'anon_execute', has_function_privilege('anon', p.oid, 'EXECUTE'),
             'authenticated_execute', has_function_privilege('authenticated', p.oid, 'EXECUTE'),
             'service_role_execute', has_function_privilege('service_role', p.oid, 'EXECUTE'))
      INTO v_row
      FROM pg_proc p
     WHERE p.oid = to_regprocedure('public.' || v_f || '()');
    v_fn := v_fn || jsonb_build_object(v_f, COALESCE(v_row, jsonb_build_object('exists', false)));
  END LOOP;
  RETURN jsonb_build_object('tables', v_tbl, 'functions', v_fn);
END;
$$;
