    98	  $PSQL -X -e -d "$db" -f "$WORK/$name.expanded.sql" "$@" > "$o" 2>&1
    99	  cat "$o" >> "$T"
   100	  local pass fail
   101	  pass=$(grep -c '^PASS:' "$o"); fail=$(grep -c '^FAIL:' "$o")
   102	  # N-2: the expected assertion count is DECLARED in the source (-- EXPECT_PASS: N) and cross-checked against a static
   103	  # count of the source's own assertion lines outside \if blocks (the RED-only controls live inside \if).
   104	  SCN_EXPECT=$(grep -m1 -oE '^-- EXPECT_PASS: [0-9]+' "$SRC/$name.sql" | grep -oE '[0-9]+$')
   105	  SCN_STATIC=$(awk 'BEGIN{d=0} /^\\if /{d++} /^\\endif/{d--} d==0 && /^SELECT t\.(expect|val)\(/{n++} END{print n+0}' "$SRC/$name.sql")
   106	  p ""; p ">>> RESULT $db: PASS=$pass FAIL=$fail   (declared EXPECT_PASS=${SCN_EXPECT:-none}, static assertion lines=$SCN_STATIC)"
   107	  SCN_PASS=$pass; SCN_FAIL=$fail
   108	}

   110	# ---- mutants (each is NAMED; a RED run applies exactly one) ---------------------------------------------------
   111	mut_file() {  # $1 name -> writes $WORK/mutant-$1.sql ; prints a diff into the transcript
   112	  local n=$1 f=$WORK/mutant-$1.sql
   113	  case $n in
   114	    no_guard)       sed "s/IF current_setting('transaction_isolation') <> 'read committed' THEN/IF false THEN/" "$SQLF" > "$f";;
   115	    no_for_share)   sed "s/^     FOR SHARE;\$/     ;/" "$SQLF" > "$f";;
   116	    old_order)      sed "s/^     AND NEW.created_from_invoice_id IS NOT NULL THEN\$/     AND false THEN/" "$SQLF" > "$f";;
   117	    no_tr4b)      sed "s/IF NEW.delivery_type = 'internal_transfer' AND OLD.delivery_type IS DISTINCT FROM 'internal_transfer' THEN/IF false THEN/" "$SQLF" > "$f";;
   118	    no_b2_insert)   sed -e "s/^  IF NEW.unlinked_at IS NOT NULL OR NEW.unlinked_by IS NOT NULL OR NEW.unlink_reason IS NOT NULL\$/  IF (NEW.unlinked_at IS NOT NULL OR NEW.unlinked_by IS NOT NULL OR NEW.unlink_reason IS NOT NULL/" -e "s/^     OR NEW.superseded_approval_at IS NOT NULL THEN\$/     OR NEW.superseded_approval_at IS NOT NULL) AND false THEN/" "$SQLF" > "$f";;
   119	    rls_no_status)  sed "/AND u.status = 'active'/d" "$SQLF" > "$f";;
   120	    rls_no_role)    sed "/AND pm.role = 'pm'/d" "$SQLF" > "$f";;
   121	    *) echo "unknown file mutant $n" >&2; return 1;;
   122	  esac
   123	  p "MUTANT (named): $n   -- diff of the real file against the mutated copy:"
   124	  diff "$SQLF" "$f" >> "$T"; true
   125	}

   138	# generic scenario test: RED list, then GREEN
   139	generic() {
   140	  local scenario=$1; shift
   141	  local reds=("$@")
   142	  header
   143	  local n=0 red_ok=1
   144	  for r in "${reds[@]}"; do
   145	    n=$((n+1))
   146	    p "################ RED run $n: $r ################"
   147	    local kind=${r%%:*} name=${r#*:} db="t_${ID//-/_}_red$n"
   148	    case $kind in
   149	      file) mut_file "$name" || exit 1; mkdb "$db" "$WORK/mutant-$name.sql";;
   150	      hfile) HELPER_FILE=$WORK/mutant-helper-$name.sql
   151	             case $name in no_guard) sed "s/^  SELECT p_tenant_id IS NOT NULL AND EXISTS (\$/  SELECT true OR p_tenant_id IS NOT NULL AND EXISTS (/" "$HELPF" > "$HELPER_FILE";; esac
   152	             p "MUTANT (named): helper $name   -- diff of the real helper script against the mutated copy:"; diff "$HELPF" "$HELPER_FILE" >> "$T"
   153	             mkdb "$db" "$SQLF"; HELPER_FILE=$HELPF;;
   154	      commit) git -C "$ROOT" show "$name:docs/reviews/049_capture_tables.sql" > "$WORK/old-$name.sql"
   155	              p "MUTANT (named): the REVIEWED file at commit $name   -- diff of that file against the file under test:"; diff "$WORK/old-$name.sql" "$SQLF" >> "$T"
   156	              mkdb "$db" "$WORK/old-$name.sql";;
   157	      post) p "MUTANT (named): $name   (post-UP SQL applied to the real file's result)"; mkdb "$db" "$SQLF" "$(mut_post "$name")";;
   158	      flag) p "MUTANT (named): flow variant -v $name"; mkdb "$db" "$SQLF";;
   159	    esac
   160	    if [ "$kind" = flag ]; then scn "$db" "$scenario" -v "$name"; else scn "$db" "$scenario"; fi
   161	    if [ "${SCN_FAIL:-0}" -gt 0 ]; then p ">>> RED verdict: test FAILED against the mutant (FAIL=$SCN_FAIL) -- the test can fail: OK"; else p ">>> RED verdict: test did NOT fail against the mutant -- NOT A VALID TEST"; red_ok=0; fi
   162	    p ""
   163	  done
   164	  p "################ GREEN run: the real file ################"
   165	  local gdb="t_${ID//-/_}_green"
   166	  mkdb "$gdb" "$SQLF"
   167	  if [ "${#reds[@]}" -gt 0 ] && [[ ${reds[0]} == flag:* ]]; then scn "$gdb" "$scenario" -v green=1; else scn "$gdb" "$scenario"; fi
   168	  # N-2 gate: PASS must EQUAL the declared expected count (which must equal the static count); FAIL must be 0.
   169	  if [ "${SCN_FAIL:-1}" -eq 0 ] && [ -n "${SCN_EXPECT:-}" ] && [ "${SCN_PASS:-0}" -eq "$SCN_EXPECT" ] && [ "$SCN_EXPECT" -eq "$SCN_STATIC" ] && [ "$red_ok" -eq 1 ]; then
   170	    p ">>> OVERALL $ID: RED failed as required, GREEN PASS=$SCN_PASS == expected $SCN_EXPECT (static $SCN_STATIC), FAIL=0"; return 0
   171	  fi
   172	  p ">>> OVERALL $ID: NOT CLEAN (red_ok=$red_ok green PASS=${SCN_PASS:-0} expected=${SCN_EXPECT:-none} static=${SCN_STATIC:-?} FAIL=${SCN_FAIL:-?})"; return 1
   173	}

   175	# ---- custom driver: T-B3, link-vs-retype races with REAL concurrent psql sessions --------------------------------
   176	# These are separate psql processes against a local PG 17 cluster: genuinely concurrent. (The CLAUDE.md sandbox limit
   177	# -- serialised concurrent RPC calls -- concerns test-db calls through the JS client only; it does not apply here.)
   178	# They prove the trigger design on PG 17; the PR-B functions' own race tests remain CI-only.
   179	expand() { python3 -I "$SRC/expand.py" "$SRC/$1.sql" "$WORK/$1.expanded.sql"; }
   180	race() {  # label first second exp_type exp_links [red-spec]
   181	  local label=$1 first=$2 second=$3 et=$4 el=$5 spec=${6:-}
   182	  local db="t_b3_${label}_$([ -n "$spec" ] && echo red || echo green)"
   183	  if [ -n "$spec" ]; then
   184	    local name=${spec#file:}
   185	    p "################ RED race $label: MUTANT (named) $name ################"
   186	    mut_file "$name" || exit 1; mkdb "$db" "$WORK/mutant-$name.sql"
   187	  else
   188	    p "################ GREEN race $label: the real file ################"
   189	    mkdb "$db" "$SQLF"
   190	  fi
   191	  for f in B3-setup B3-observe B3-end "$first" "$second"; do expand "$f" || exit 1; done
   192	  p "\$ psql -X -e -d $db -f B3-setup.sql"; $PSQL -X -e -d "$db" -f "$WORK/B3-setup.expanded.sql" >> "$T" 2>&1
   193	  p "\$ [session 1] psql -X -e -d $db -f $first.sql   (background, starts at T+0.0s)"
   194	  $PSQL -X -e -d "$db" -f "$WORK/$first.expanded.sql" > "$WORK/s1.out" 2>&1 &
   195	  local p1=$!
   196	  sleep 1.5
   197	  p "\$ [session 2] psql -X -e -d $db -f $second.sql   (background, starts at T+1.5s)"
   198	  $PSQL -X -e -d "$db" -f "$WORK/$second.expanded.sql" > "$WORK/s2.out" 2>&1 &
   199	  local p2=$!
   200	  sleep 1.5
   201	  p "\$ [observer] psql -X -d postgres -v db=$db -f B3-observe.sql   (at T+3.0s, both writers open)"
   202	  $PSQL -X -d postgres -v db="$db" -f "$WORK/B3-observe.expanded.sql" >> "$T" 2>&1
   203	  wait $p1 $p2
   204	  p "----- session 1 output ($first) -----"; cat "$WORK/s1.out" >> "$T"
   205	  p "----- session 2 output ($second) -----"; cat "$WORK/s2.out" >> "$T"
   206	  p "\$ psql -X -e -d $db -v exp_type=$et -v exp_links=$el -f B3-end.sql"
   207	  $PSQL -X -e -d "$db" -v exp_type="$et" -v exp_links="$el" -f "$WORK/B3-end.expanded.sql" > "$WORK/end.out" 2>&1; cat "$WORK/end.out" >> "$T"
   208	  RACE_FAIL=$(cat "$WORK/s1.out" "$WORK/s2.out" "$WORK/end.out" | grep -c '^FAIL:')
   209	  RACE_PASS=$(cat "$WORK/s1.out" "$WORK/s2.out" "$WORK/end.out" | grep -c '^PASS:')
   210	  p ">>> RESULT race $label ($([ -n "$spec" ] && echo RED || echo GREEN)): PASS=$RACE_PASS FAIL=$RACE_FAIL"
   211	}
   212	b3() {
   213	  header
   214	  p "NOTE: three races, each RED (named mutant) then GREEN (real file). Two writers are separate psql processes, genuinely"
   215	  p "concurrent on a local PG 17 cluster; 'LINK' = pm_manual link INSERT (TR1/TR2), 'RETYPE' = UPDATE delivery_type (TR4b)."
   216	  local ok=1
   217	  race rc_link_first   B3a-link    B3a-retype  supplier 1 file:no_tr4b;     [ "$RACE_FAIL" -gt 0 ] || { p ">>> RED did NOT fail -- not a valid test"; ok=0; }
   218	  race rc_link_first   B3a-link    B3a-retype  supplier 1;                  [ "$RACE_FAIL" -eq 0 ] && [ "$RACE_PASS" -eq 4 ] || ok=0
   219	  race rc_retype_first B3b-retype  B3b-link    internal_transfer 0 file:no_for_share; [ "$RACE_FAIL" -gt 0 ] || { p ">>> RED did NOT fail -- not a valid test"; ok=0; }
   220	  race rc_retype_first B3b-retype  B3b-link    internal_transfer 0;         [ "$RACE_FAIL" -eq 0 ] && [ "$RACE_PASS" -eq 4 ] || ok=0
   221	  race rr              B3c-retype  B3c-link    supplier 1 file:no_guard;    [ "$RACE_FAIL" -gt 0 ] || { p ">>> RED did NOT fail -- not a valid test"; ok=0; }
   222	  race rr              B3c-retype  B3c-link    supplier 1;                  [ "$RACE_FAIL" -eq 0 ] && [ "$RACE_PASS" -eq 4 ] || ok=0
   223	  if [ $ok -eq 1 ]; then p ">>> OVERALL T-B3: every RED failed as required, every GREEN clean"; else p ">>> OVERALL T-B3: NOT CLEAN"; return 1; fi
   224	}

#!/usr/bin/env python3
"""Expand fixture-ID tokens in a scenario file to full UUID literals (so the transcript shows what ran).
Tokens: {T1} tenant, {P1} project, {U.PM1} users.id, {A.PM1} users.auth_id, {D1} document, {V1} delivery, {L1} link.
usage: expand.py <in.sql> <out.sql>
"""
import re, sys

USERS = {'PM1': 1, 'ENG1': 2, 'QS1': 3, 'DPM': 4, 'ADMN': 5, 'ADMP': 6, 'PM2': 7, 'PEND': 8, 'PM3': 9}

def uuid(prefix: int, n: int) -> str:
    return f'{prefix}0000000-0000-0000-0000-{n:012d}'

def sub(m):
    kind, arg = m.group(1), m.group(2)
    if kind == 'T': return uuid(1, int(arg))
    if kind == 'P': return uuid(2, int(arg))
    if kind == 'U.': return uuid(3, USERS[arg])
    if kind == 'A.': return uuid(4, USERS[arg])
    if kind == 'D': return uuid(5, int(arg))
    if kind == 'V': return uuid(6, int(arg))
    if kind == 'L': return uuid(7, int(arg))
    raise SystemExit(f'bad token {m.group(0)}')

text = open(sys.argv[1]).read()
out = re.sub(r'\{(T|P|U\.|A\.|D|V|L)([A-Za-z0-9]+)\}', sub, text)
if re.search(r'\{[TPUADVL]\.?[A-Za-z0-9]+\}', out):
    raise SystemExit('unexpanded token left')
open(sys.argv[2], 'w').write(out)
