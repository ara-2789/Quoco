const { createClient } = require('@supabase/supabase-js')
const c = createClient(process.env.SUPABASE_TEST_URL, process.env.SUPABASE_TEST_SERVICE_ROLE_KEY, { auth: { persistSession: false } })
console.log('host:', new URL(process.env.SUPABASE_TEST_URL).hostname)
console.log('Bodies are zero bytes; the bucket checks the declared content type only, not file content. The bucket is a backstop; PR-C byte checks decide acceptance (049 lines 1013-1018).')
const run = 'zz-2b-bkt-' + Date.now()
console.log('prefix:', run)
const MiB = 1024 * 1024
const MIME = /mime/i
const SIZE = /size|exceed|too large/i
const cases = [
  { id: 'T-BKT-1', name: 'text/plain refused',          path: `${run}/a.txt`,   type: 'text/plain',      bytes: 16,           expect: 'refused',  reason: MIME },
  { id: 'T-BKT-2', name: '16 MiB + 1 byte PDF refused', path: `${run}/big.pdf`, type: 'application/pdf', bytes: 16 * MiB + 1, expect: 'refused',  reason: SIZE },
  { id: 'T-BKT-3', name: 'HEIC refused (D-9)',          path: `${run}/a.heic`,  type: 'image/heic',      bytes: 16,           expect: 'refused',  reason: MIME },
  { id: 'T-BKT-4', name: '16 MiB PDF accepted',         path: `${run}/max.pdf`, type: 'application/pdf', bytes: 16 * MiB,     expect: 'accepted' },
  { id: 'T-BKT-5', name: 'small JPEG accepted',         path: `${run}/a.jpg`,   type: 'image/jpeg',      bytes: 16,           expect: 'accepted' },
]
const accepted = []
let ok = 0
for (const k of cases) {
  const body = new Blob([new Uint8Array(k.bytes)], { type: k.type })
  const { data, error } = await c.storage.from('capture-documents').upload(k.path, body, { contentType: k.type })
  const got = error ? 'refused' : 'accepted'
  if (!error) accepted.push(k.path)
  let pass = got === k.expect
  let why = ''
  if (error) {
    const msg = String(error.message ?? '')
    console.log(`  error detail: statusCode=${error.statusCode ?? error.status ?? '?'} name=${error.name ?? '?'} code=${error.error ?? error.code ?? '?'} message=${msg.slice(0, 200)}`)
    if (pass && k.reason && !k.reason.test(msg)) { pass = false; why = ` (refused for the wrong reason: message does not match ${k.reason})` }
  }
  if (pass) ok++
  console.log(`${pass ? 'PASS' : 'FAIL'}: ${k.id} ${k.name} -- expected ${k.expect}, got ${got}${why}` + (error ? '' : ` | path=${data?.path}`))
}
console.log(`RESULT: PASS=${ok} FAIL=${cases.length - ok}`)
console.log('ACCEPTED_PATHS:', JSON.stringify(accepted))
